    <script>
        // Festival Configuration
        const FESTIVAL_CONFIG = {
            // Set this to true to enable festival effects by default
            ENABLED_BY_DEFAULT: true,  // Change to false if you want it off by default

            // Color scheme for current festival
            COLORS: [
                '#f59e0b', // Amber
                '#c026d3', // Pink
                '#6d28d9', // Purple
                '#06b6d4', // Cyan
                '#ec4899', // Hot Pink
                '#8b5cf6'  // Violet
            ]
        };

        // Festive Effects Manager
        class FestiveEffects {
            constructor() {
                this.isActive = FESTIVAL_CONFIG.ENABLED_BY_DEFAULT;
                this.container = document.getElementById('festive-effects');
                this.toggleBtn = document.getElementById('festivalToggle');
                this.badge = document.querySelector('.festival-badge');

                this.init();
            }

            init() {
                // Set initial state
                if (this.isActive) {
                    this.enableEffects();
                    this.toggleBtn.classList.add('active');
                    this.badge.textContent = 'ON';
                }

                // Add toggle listener
                this.toggleBtn.addEventListener('click', () => this.toggle());
            }

            toggle() {
                this.isActive = !this.isActive;

                if (this.isActive) {
                    this.enableEffects();
                    this.toggleBtn.classList.add('active');
                    this.badge.textContent = 'ON';
                } else {
                    this.disableEffects();
                    this.toggleBtn.classList.remove('active');
                    this.badge.textContent = 'OFF';
                }
            }

            enableEffects() {
                if (!this.isActive) return;

                this.container.style.display = 'block';
                document.body.classList.add('festive-active');

                // Create different effects
                this.createColorOverlay();
                this.createColorOrbs();
                this.createLightRays();
                this.createParticles();
            }

            disableEffects() {
                this.container.style.display = 'none';
                document.body.classList.remove('festive-active');

                // Clear all effects
                this.container.innerHTML = '';
            }

            createColorOverlay() {
                const overlay = document.createElement('div');
                overlay.className = 'color-overlay';
                this.container.appendChild(overlay);
            }

            createColorOrbs() {
                const positions = [
                    { top: '10%', left: '5%' },
                    { top: '20%', right: '10%' },
                    { bottom: '15%', left: '15%' },
                    { bottom: '25%', right: '5%' },
                    { top: '50%', left: '20%' },
                    { top: '70%', right: '15%' },
                    { bottom: '40%', left: '30%' },
                    { bottom: '60%', right: '25%' }
                ];

                positions.forEach((pos, index) => {
                    const orb = document.createElement('div');
                    orb.className = 'color-orb';
                    orb.style.background = `radial-gradient(circle, ${FESTIVAL_CONFIG.COLORS[index % FESTIVAL_CONFIG.COLORS.length]} 0%, transparent 70%)`;
                    orb.style.animationDelay = `${index * 2}s`;

                    Object.assign(orb.style, pos);
                    this.container.appendChild(orb);
                });
            }

            createLightRays() {
                const rays = document.createElement('div');
                rays.className = 'light-ray';
                this.container.appendChild(rays);
            }

            createParticles() {
                const createParticle = () => {
                    const particle = document.createElement('div');
                    particle.className = 'color-particle';

                    particle.style.background = FESTIVAL_CONFIG.COLORS[Math.floor(Math.random() * FESTIVAL_CONFIG.COLORS.length)];
                    particle.style.left = Math.random() * 100 + '%';
                    particle.style.animationDuration = (10 + Math.random() * 10) + 's';
                    particle.style.animationDelay = Math.random() * 5 + 's';
                    particle.style.width = (2 + Math.random() * 6) + 'px';
                    particle.style.height = particle.style.width;

                    this.container.appendChild(particle);

                    // Remove after animation
                    setTimeout(() => particle.remove(), 20000);
                };

                // Create initial particles
                for (let i = 0; i < 30; i++) {
                    setTimeout(() => createParticle(), i * 100);
                }

                // Continuous particles
                const particleInterval = setInterval(() => {
                    if (this.isActive) {
                        createParticle();
                    } else {
                        clearInterval(particleInterval);
                    }
                }, 500);
            }
        }

        // Initialize effects when DOM is loaded
        document.addEventListener('DOMContentLoaded', () => {
            window.festiveEffects = new FestiveEffects();
        });
    </script>
